\begin{frame}{Research Context} \small
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{itemize}
        \item \textbf{ML in non-technical domains}:\\
          Domain experts need ML without requiring data science training

        \item \textbf{AutoML democratization}:\\
          Performance-first AutoML neglects transparency, user control and
          interpretability of results

        \item \textbf{HCAI paradigm}:\\
          AI should amplify human capabilities, not replace them ---
          yet few tools have been systematically evaluated under HCAI criteria

        \item \textbf{Gap in HCI evaluation}:\\
          Existing heuristic inspection instruments are limited to
          classical usability criteria and do \emph{not} incorporate
          HCAI dimensions
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Two Interrelated Contributions}
        \begin{enumerate} \scriptsize
          \item Extension of heuristic inspection methods to HGML tools
          via the \textbf{HCAI-HGML Compliance Matrix}
          \item Illustrative case study applying the instrument
          to \textbf{InsuML}, an HGML tool for non-expert users
        \end{enumerate}
      \end{block}
      \vspace{4mm}
      \begin{exampleblock}{Core Research Question} \scriptsize
        Which criteria determine whether a pipeline decision genuinely
        belongs to the user or to the system, and how can the
        resulting human-automation balance be characterized in a
        reproducible and comparable manner?
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Background}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Human-Centered Artificial Intelligence (HCAI)} \small
  \begin{minipage}{0.5\textwidth}
    \begin{block}{HCAI Framework (Shneiderman, 2022)}
      \begin{itemize}
        \item \textbf{Paradigm shift}:\\
          AI should amplifies human capabilities rather than replacing them
        \item \textbf{Two-dimensional conception}:\\
          High automation \emph{and} high human control are
          simultaneously achievable
      \end{itemize}
    \end{block}
    \vspace{3mm}
    \textbf{Human control criteria}:
    ease of use · pipeline configurability · information interpretability\\[2mm]
    \textbf{Automation criteria}:
    data type detection · data cleaning · feature engineering ·
    model training · hyperparameter tuning · model validation
  \end{minipage}%
  \begin{minipage}{0.5\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.65, transform shape]

  \end{minipage}
\end{frame}

\begin{frame}{Human-Guided Machine Learning (HGML)} \small
  \begin{block}{HGML Concept (Gil et al., 2019)}
    Practical realization of HCAI principles within the ML domain:
    active participation of domain experts throughout the pipeline
    enriches the process with contextual knowledge that algorithms
    alone cannot incorporate
  \end{block}
  \vspace{3mm}
  \begin{block}{Three Pipeline Areas}
    \begin{itemize}\scriptsize
      \item \textbf{Data Use}: variable selection, variable requirements,
            instance selection --- \emph{users intervene in variable
            selection/transformation, subpopulation definition and
            outlier management}
      \item \textbf{Model Development}: model selection, parameter settings,
            solution design --- \emph{experts guide algorithm family
            selection, impose parameter constraints and influence
            pipeline design}
      \item \textbf{Model Interpretation}: model assessment, model comparison,
            parameter comparison --- \emph{users request specific metrics,
            compare configurations and analyze sensitivity to parameters}
    \end{itemize}
  \end{block}
  \vspace{2mm}
  \begin{alertblock}{Limitation in Practice}\scriptsize
    Few HGML proposals offer complete software implementations
    accessible to users without AI training
  \end{alertblock}
\end{frame}

\begin{frame}{Heuristic Inspection Methods in HCI} \small
  \begin{block}{What They Are}
    \begin{itemize}\scriptsize
      \item Expert-based evaluation \textbf{without user participation}
      \item Typically \textbf{3--5 domain experts} analyze an interface
            against predefined principles
      \item Suitable for early-stage evaluation, iterative design cycles and contexts where user recruitment is impractical
    \end{itemize}
  \end{block}
  \vspace{2mm}
  \begin{block}{Two Established Techniques}
    \begin{itemize}\scriptsize
      \item \textbf{Heuristic evaluation with severity rating}:
            identify usability problems and classify each by severity
            on an ordinal scale, to prioritize redesign efforts
      \item \textbf{Scored heuristic checklist}: transforms usability
            principles into a list of evaluable criteria, each scored
            quantitatively (Likert-type scale), yielding aggregate
            compliance measures
    \end{itemize}
  \end{block}
  \vspace{2mm}
  \begin{alertblock}{Limitation}\scriptsize
    Existing instruments cover only classical usability criteria
    (e.g. visibility of system status, consistency) and do \textbf{not}
    incorporate HCAI dimensions
  \end{alertblock}
\end{frame}

\section{The HCAI-HGML Inspection Instrument}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Dimensions and Criteria} \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Human Control --- 3 Criteria}
        \begin{itemize}\scriptsize
          \item \textbf{Ease of use}: operable without AI expertise;
                no programming required
          \item \textbf{Pipeline configurability}: user can customize
                key decisions across phases without code intervention,
                injecting domain knowledge
          \item \textbf{Information interpretability}: system communicates
                its decisions and results in formats accessible to
                non-technical users
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Computer Automation --- 6 Criteria}
        \begin{itemize}\scriptsize
          \item \textbf{Data type detection}: automatic identification of
                variable types
          \item \textbf{Data cleaning}: detection and resolution of quality
                issues (missing values, duplicates)
          \item \textbf{Feature engineering}: generation, transformation
                or selection of features
          \item \textbf{Model training}: architecture instantiation and
                optimization without manual intervention
          \item \textbf{Hyperparameter tuning}: automatic search of
                optimal configurations
          \item \textbf{Model validation}: cross-validation or hold-out
                evaluation and performance reporting
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \begin{exampleblock}{Key Design Decision} \scriptsize
    The six automation sub-criteria are collapsed into a single
    $A_{\text{global}}$ per phase to avoid penalizing tools for
    sub-criteria that are structurally inapplicable to a given stage.
  \end{exampleblock}
\end{frame}

\begin{frame}{Theoretical Responsibility Distribution} \small
  \begin{table}
    \centering
    \renewcommand{\arraystretch}{1.2}
    \resizebox{\linewidth}{!}{%
    \begin{tabular}{l p{5.5cm} p{5.5cm}}
      \toprule
      \textbf{Pipeline Phase} & \textbf{Human Control (expected)} & \textbf{Computer Automation (expected)} \\
      \midrule
      \multicolumn{3}{l}{\textbf{Data Use}} \\
      \midrule
      Variable selection    & User prioritizes variables, proposes combinations, identifies missing data
                            & System infers relevant features and performs automatic imputation \\
      Variable requirements & Expert specifies inclusion/exclusion based on domain knowledge
                            & System enforces rules and validates dataset consistency \\
      Instance selection    & User defines subpopulations, removes outliers, sets splitting strategy
                            & System performs automatic splitting, balancing and cross-validation \\
      \midrule
      \multicolumn{3}{l}{\textbf{Model Development}} \\
      \midrule
      Model selection       & User indicates preferred model types
                            & System builds and evaluates candidate architectures \\
      Parameter settings    & Expert specifies values informed by domain knowledge
                            & System searches and optimizes hyperparameters within declared ranges \\
      Solution design       & User modifies preprocessing, feature generation and pipeline design
                            & System constructs complete pipelines (encoders, combiners, preprocessing) \\
      \midrule
      \multicolumn{3}{l}{\textbf{Model Interpretation}} \\
      \midrule
      Model assessment      & User requests specific metrics
                            & System computes metrics, training curves and diagnostics \\
      Model comparison      & User selects models or configurations to compare
                            & System manages experiments and provides comparative visualizations \\
      Parameter comparison  & User identifies parameters for targeted variation
                            & System runs ablations and reports performance impact \\
      \bottomrule
    \end{tabular}}
  \end{table}
\end{frame}

\begin{frame}{The HCAI-HGML Compliance Matrix} \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Structure}
        \begin{itemize}\scriptsize
          \item \textbf{Rows}: 9 pipeline phases across Data Use,
                Model Development and Model Interpretation
          \item \textbf{Columns}: 3 human control criteria
                + 1 $A_{\text{global}}$ per phase
          \item \textbf{Scale}: five-level ordinal
                \begin{itemize}
                  \item 0.00 --- Absent
                  \item 0.25 --- Minimal (rudimentary, severe limitations)
                  \item 0.50 --- Partial ($\approx$ half the expected behavior)
                  \item 0.75 --- Substantial (largely met, minor gaps)
                  \item 1.00 --- Complete (fully satisfied)
                \end{itemize}
        \end{itemize}
      \end{block}
      \vspace{2mm}
      \begin{exampleblock}{Aggregate Scores}\scriptsize
        \[
          S_{HC} = \frac{1}{n \cdot c_{HC}}\sum_{i=1}^{n}\sum_{j=1}^{c_{HC}} v_{ij}
          \qquad
          S_{A} = \frac{1}{n}\sum_{i=1}^{n} A_{\text{global},i}
        \]
        $n=9$ phases, $c_{HC}=3$ human control criteria
      \end{exampleblock}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{HCAI Quadrant Classification}
        \begin{tabular}{p{3.0cm} l l}
          \toprule
          \textbf{Quadrant} & $S_{HC}$ & $S_{A}$ \\
          \midrule
          \cellcolor{green!20}\textbf{Optimal} & $>0.5$ & $>0.5$ \\
          Automated but opaque & $\leq0.5$ & $>0.5$ \\
          Controlled but manual & $>0.5$ & $\leq0.5$ \\
          Deficient & $\leq0.5$ & $\leq0.5$ \\
          \bottomrule
        \end{tabular}
      \end{block}
      \vspace{3mm}
      \begin{block}{Distance to Ideal Point}\scriptsize
        \[
          D = \sqrt{(1-S_{HC})^2 + (1-S_{A})^2}
        \]
        $D \in [0,\,\sqrt{2}]$.
        A low $D$ within the optimal quadrant signals proximity
        to full HCAI compliance.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Case Study: InsuML}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{InsuML --- Tool Overview} \small
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{What is InsuML?}
        \begin{itemize}\scriptsize
          \item HGML software application under active development
          \item Designed for users \textbf{without AI expertise}
          \item Combines an AutoML engine with a \textbf{guided GUI}
                that makes every pipeline decision visible and controllable
          \item No programming required at any stage
        \end{itemize}
      \end{block}
      \vspace{2mm}
      \begin{block}{Distinctive Interaction Features}
        \begin{itemize}\scriptsize
          \item \textbf{Tab-based navigation}: sequential, bidirectional ---
                forward requires completing current tab; backward is always permitted
          \item \textbf{Pre-training configuration summary}:
                all user-selected options and automatic defaults displayed
                before execution begins
          \item \textbf{Progressive disclosure}: each tab exposes
                only the decisions relevant to its pipeline phase
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Three-Layer Architecture}
        \begin{itemize}\scriptsize
          \item \textbf{Interface Layer} (Python/Tkinter):\\
                manages user interaction and input validation
          \item \textbf{Core Logic Layer}:\\
                translates user selections into technical configurations;
                generates Ludwig config files automatically
          \item \textbf{AutoML Layer} (Ludwig engine):\\
                handles training, hyperparameter optimization
                and model evaluation autonomously
        \end{itemize}
      \end{block}
      \vspace{2mm}
      \begin{exampleblock}{Design Premise}\scriptsize
        Domain knowledge and expert intuition are valuable assets
        that must be \emph{explicitly integrated} into the ML process,
        rather than displaced by automation.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Discussion}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Design Principles for ML Interfaces} \small
  Three reusable principles emerging from the inspection:
  \vspace{3mm}

  \begin{block}{1 · Sequential Tab-Based Navigation}
    \scriptsize
    Structures interaction as a guided progression through pipeline phases,
    exposing only relevant decisions at each step, with bidirectional access
    to revisit and revise earlier choices.
    \textbf{Effect}: ease of use scores of 1.00 across six of nine phases ---
    a reliable interaction model for complex, multi-step ML workflows.
  \end{block}

  \begin{block}{2 · Explicit Responsibility Allocation}
    \scriptsize
    Making visible which decisions belong to the user and which are handled
    automatically. The \textbf{pre-training configuration summary} exemplifies this:
    every pending automated decision becomes an inspectable checkpoint that
    the user can review and correct before training begins.
    \textbf{Effect}: directly supports information interpretability scores
    of 0.75--1.00 across most phases.
  \end{block}

  \begin{block}{3 · Modular Architectural Separation}
    \scriptsize
    Decoupling interface, core logic and AutoML layers keeps the translation
    between user intentions and technical parameters explicit and traceable.
    The AutoML backend can evolve independently without disrupting the
    interaction model. \textbf{Effect}: contributes to bias mitigation and
    responsible AI deployment.
  \end{block}
\end{frame}

\begin{frame}{Limitations and Future Work} \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Current Limitations}
        \begin{itemize}\scriptsize
          \item \textbf{Single evaluator}: scores assigned by the tool's
                own authors --- independent auditors required for formal
                validity
          \item \textbf{No empirical usability studies}: task completion
                time, error rates and user satisfaction not yet measured
                with real domain experts
          \item \textbf{Limited explainability}: only global performance
                metrics and confusion matrices; no XAI techniques (SHAP,
                LIME) for individual prediction explanations
          \item \textbf{Missing parameter comparison}: zero-scoring phase
                that limits systematic experimentation
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Roadmap}
        \begin{enumerate}\scriptsize
          \item \textbf{Independent reliability study}:
                3--5 domain experts apply the matrix to InsuML ---
                essential for consolidating the instrument
          \item \textbf{Empirical usability studies}:
                HCI research methods with real specialists performing
                realistic modeling tasks
          \item \textbf{XAI integration}:
                SHAP and LIME for feature-level explanation of
                individual predictions
          \item \textbf{Parameter comparison}:
                implement ablation/sensitivity module ---
                concrete development priority
          \item \textbf{Heterogeneous data sources}:
                extend toward text, images and time series while
                maintaining human-centered design principles
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusions}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Conclusions} \scriptsize
  \begin{block}{Main Contributions}
    \begin{itemize}
      \item \textbf{Novel inspection instrument}: HCAI-HGML Compliance Matrix
            extends scored heuristic checklists with HCAI dimensions across
            nine HGML pipeline phases --- without requiring user participation
      \item \textbf{Formal scoring}: aggregate measures $S_{HC}$, $S_{A}$
            and distance metric $D$ yield a quantifiable, comparable HCAI
            profile for any HGML tool
      \item \textbf{Validated through case study}: InsuML achieves
            $S_{HC} = 0.64$, $S_{A} = 0.86$, $D = 0.39$ --- firmly in
            the Optimal quadrant --- with primary gaps localized in model
            development configurability and parameter comparison
      \item \textbf{Reusable design vocabulary}: sequential guided navigation,
            explicit responsibility allocation and modular architectural
            separation constitute transferable principles for any ML tool
            targeting non-expert users
    \end{itemize}
  \end{block}
  \vspace{2mm}
  \begin{block}{Take-Home Messages}
    \begin{itemize}
      \item Existing heuristic inspection instruments \textbf{cannot assess HCAI
            compliance} --- a gap that grows as AI components permeate
            interactive systems.
      \item High automation and meaningful human control are \textbf{not
            trade-offs} when interface design places the non-expert user
            at the center.
      \item Interface design for ML systems is a \textbf{determining factor}
            for effective and responsible human-AI collaboration.
    \end{itemize}
  \end{block}
\end{frame}
