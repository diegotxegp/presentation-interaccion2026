\begin{frame}{Research Context} \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{ML in non-technical domains}:\\
          Domain experts need ML without requiring data science knowledge

        \item \textbf{AutoML democratization}:\\
          Performance-first AutoML neglects transparency, user control and
          interpretability of results

        \item \textbf{Human-Centered AI (HCAI) --- the user at the center}:\\
          An active participant with control, not a passive
          recipient --- yet few AutoML tools evaluated under HCAI criteria

        \item \textbf{Gap in Human-Computer Interaction (HCI)}:\\
          Existing heuristic inspection instruments are limited to
          classical usability criteria and do \emph{not} incorporate
          HCAI criteria

          \item Extension of heuristic inspection methods

          \item Case study applying the instrument

\section{Background}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Human-Centered Artificial Intelligence (HCAI)} \small
  \begin{minipage}{0.5\textwidth}
    \begin{block}{HCAI Framework (Shneiderman, 2022)}
      \begin{itemize}
        \item \textbf{Paradigm shift}:\\
          AI should amplify human capabilities rather than replacing them
        \item \textbf{Two-dimensional conception}:\\
          High automation \emph{and} high human control are
          simultaneously achievable
      \end{itemize}
    \end{block}
    \vspace{3mm}
    \textbf{Human control criteria}:
    ease of use · pipeline configurability · information interpretability\\[2mm]
    \textbf{Automation criteria}:
    data type detection · data cleaning · feature engineering ·
    model training · hyperparameter tuning · model validation
  \end{minipage}%
  \begin{minipage}{0.5\textwidth}
    \centering
    \begin{tikzpicture}[scale=0.65, transform shape]
      \draw[->, thick] (0,0) -- (8,0)
        node[anchor=north, xshift=-1.5cm, yshift=-0.2cm] {Computer Automation};
      \draw[->, thick] (0,0) -- (0,8)
        node[anchor=south east, rotate=90, xshift=-0.2cm, yshift=0.2cm] {Human Control};
      \draw[dashed] (0,4) -- (8,4);
      \draw[dashed] (4,0) -- (4,8);
      \node[align=center, font=\scriptsize] at (2,2)
        {Low \\ Human Control \\ Low \\ Automation};

        {High \\ Human Control \\ Low \\ Automation};

        {\textbf{HCAI Goal} \\ High \\ Human Control \\ High \\ Automation};
    \end{tikzpicture}
  \end{minipage}
\end{frame}

\begin{frame}{Human-Guided Machine Learning (HGML)} \small
  \begin{block}{HGML Concept (Gil et al., 2019)}
    \begin{itemize}
        \item Practical realization of HCAI principles within the ML domain
        \item Active participation of domain experts throughout the pipeline
    enriches the process with contextual knowledge that algorithms
    alone cannot incorporate
    \end{itemize}
  \end{block}

\begin{frame}{Heuristic Inspection Methods in HCI} \small
  \begin{block}{What They Are}
    \begin{itemize}\scriptsize
      \item Expert-based evaluation \textbf{without user participation}
      \item Typically \textbf{3--5 domain experts} analyze an interface
            against predefined principles
      \item Suitable for early-stage evaluation, iterative design cycles and contexts where user recruitment is impractical
    \end{itemize}
  \end{block}
  \vspace{3mm}
  \centering
  \begin{tikzpicture}[scale=0.85, transform shape]

      \textbf{Heuristic Evaluation}\\\textbf{with Severity Rating}\\[1mm]
      Identify usability problems and classify each by severity

      compliance measures};

\end{frame}

\section{The HCAI-HGML Inspection Instrument}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Dimensions and Criteria} \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Human Control --- 3 Criteria}
        \begin{itemize}\scriptsize
          \item \textbf{Ease of use}: no AI expertise, no programming
          \item \textbf{Pipeline configurability}: customize decisions,
                no code intervention
          \item \textbf{Information interpretability}: accessible,
                non-technical formats
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Computer Automation --- 6 Criteria}
        \begin{itemize}\scriptsize
          \item \textbf{Data type detection}: automatic identification
          \item \textbf{Data cleaning}: missing values, duplicates
          \item \textbf{Feature engineering}: generate, transform, select
          \item \textbf{Model training}: architecture + optimization
          \item \textbf{Hyperparameter tuning}: automatic search
          \item \textbf{Model validation}: cross-validation, reporting
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \begin{exampleblock}{Key Design Decision} \scriptsize
    The six automation criteria are collapsed into a single
    $A_{\text{global}}$ per phase to avoid penalizing tools for criteria that are structurally inapplicable to a given stage.
  \end{exampleblock}
\end{frame}

\begin{frame}{Theoretical Responsibility Distribution} \small
  \centering
  \begin{tikzpicture}[scale=0.9, transform shape,
      phase/.style={rectangle, rounded corners=0.15cm, draw=UCgray,

\begin{frame}{The HCAI-HGML Compliance Matrix} \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Structure}
        \begin{itemize}\scriptsize
          \item \textbf{Rows}: 9 pipeline phases across Data Use,
                Model Development and Model Interpretation
          \item \textbf{Columns}: 3 human control criteria
                + 1 $A_{\text{global}}$ per phase
        \end{itemize}
      \end{block}
            \vspace{2mm}
      \centering
      \resizebox{0.95\columnwidth}{!}{%
      \begin{tikzpicture}[scale=1, transform shape]
        \shade[left color=red!45!white, right color=green!70!black,
               rounded corners=0.15cm] (0,0) rectangle (10,1.0);
        \foreach \x/\lbl/\name in {0/0.00/Absent, 2.5/0.25/Minimal,
              5/0.50/Partial, 7.5/0.75/Substantial, 10/1.00/Complete}
          {
            \draw[thick, white] (\x,0) -- (\x,1.0);
            \node[font=\bfseries\normalsize, anchor=south] at (\x,1.15) {\lbl};
            \node[font=\small, anchor=north, text width=2.2cm, align=center]
                  at (\x,-0.2) {\name};
          }
      \end{tikzpicture}}
      \vspace{2mm}
      \begin{exampleblock}{Aggregate Scores}\scriptsize
        \[
          S_{HC} = \frac{1}{n \cdot c_{HC}}\sum_{i=1}^{n}\sum_{j=1}^{c_{HC}} v_{ij}
          \qquad
          S_{A} = \frac{1}{n}\sum_{i=1}^{n} A_{\text{global},i}
        \]
        $n=9$ phases, $c_{HC}=3$ human control criteria
      \end{exampleblock}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{HCAI Quadrant Classification}
        \begin{tabular}{p{3.0cm} l l}
          \toprule
          \textbf{Quadrant} & $S_{HC}$ & $S_{A}$ \\
          \midrule
          \cellcolor{green!20}\textbf{Optimal} & $>0.5$ & $>0.5$ \\
          Automated but opaque & $\leq0.5$ & $>0.5$ \\
          Controlled but manual & $>0.5$ & $\leq0.5$ \\
          Deficient & $\leq0.5$ & $\leq0.5$ \\
          \bottomrule
        \end{tabular}
      \end{block}
      \vspace{3mm}
      \begin{block}{Distance to Ideal Point}\scriptsize
        \[
          D = \sqrt{(1-S_{HC})^2 + (1-S_{A})^2}
        \]
        $D \in [0,\,\sqrt{2}]$.
        A low $D$ within the optimal quadrant signals proximity
        to full HCAI compliance.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Discussion}
% ══════════════════════════════════════════════════════════════════════════════

\begin{frame}{Design Principles for ML Interfaces} \small
  Three reusable principles emerging from the inspection:
  \vspace{3mm}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \centering
      \tabicon{UCblue}\\[1mm]
      \textbf{\small Sequential\\Navigation}
      \vspace{2mm}
      \begin{block}{}
        \scriptsize
        Guided progression through pipeline phases, exposing only
        relevant decisions, with bidirectional access to revise
        earlier choices.\\[1mm]
        \textbf{Effect}: ease of use = 1.00 on 6 of 9 phases.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering
      \eyeicon{UCgreen!70!black}\\[1mm]
      \textbf{\small Explicit\\Responsibility}
      \vspace{2mm}
      \begin{block}{}
        \scriptsize
        Making visible which decisions belong to the user and which
        are automated. The pre-training summary makes every automated
        choice an inspectable checkpoint.\\[1mm]
        \textbf{Effect}: interpretability = 0.75--1.00 on most phases.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering
      \layericon{UCorange}\\[1mm]
      \textbf{\small Modular\\Architecture}
      \vspace{2mm}
      \begin{block}{}
        \scriptsize
        Decoupling interface, core logic and AutoML layers keeps
        the translation between user intent and technical parameters
        explicit and traceable.\\[1mm]
        \textbf{Effect}: bias mitigation, responsible AI deployment.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Limitations and Future Work} \small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \centering
      \magnifiericon{UCred}\\[1mm]
      \textbf{\small Current Limitations}
      \vspace{2mm}
      \begin{exampleblock}{}
        \begin{itemize}\scriptsize
          \item \textbf{Single evaluator}: scores assigned by the tool's
                own authors --- may compromise objectivity
          \item \textbf{No empirical usability studies}: completion
                time, error rates and satisfaction not yet measured
                with real domain experts
          \item \textbf{Limited scope}: explainability restricted to
                global metrics (no XAI); no parameter comparison;
                not yet tested on heterogeneous data (text, images,
                time series)
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \arrowupicon{UCgreen!70!black}\\[1mm]
      \textbf{\small Future Work}
      \vspace{2mm}
      \begin{block}{}
        \begin{enumerate}\scriptsize
          \item \textbf{Formal usability studies} with independent
                evaluators and real users
          \item \textbf{XAI integration}, \textbf{parameter comparison}
                and \textbf{heterogeneous data sources}
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
